# sum of 2 matrexies uses numpy                                                                                                        
import numpy as np

arr1 = np.array([[1, 5, 3, 2],
                [2, 1, 4, 0]])
                
print('arr1= ', arr1,  end='\n')

arr2 = np.array([[0, 4, 3, 3],
                [1, 0, 5, 7]])
                
print('arr2= ', arr2, end='\n')

print('result= ', f"{arr1} + {arr2} = {arr1 + arr2}")

result = arr1 + arr2
print('result= ', result)
